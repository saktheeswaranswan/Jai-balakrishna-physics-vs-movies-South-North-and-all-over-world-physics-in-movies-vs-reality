// === FUZZY LOGIC 5-IN-1 MULTI-SIMULATOR (HYBRID FUZZY-PID CONTROLLERS) ===
// Features: Cart-Pole, Ball-Beam, 2-Wheel Incline, 2D Drone, Gantry Crane
// Interactivity: 
// 1. Tap LEFT/RIGHT to tweak Fuzzy Sensitivity / PID Gains.
// 2. Drag center to disturb physics (add wind, shove objects).
// 3. Tap "NEXT SIM" (Top Right) to toggle between the 5 simulations.

// ========================================================
// GLOBAL APP STATE
// ========================================================
int currentSim = 0; // 0=Cart-Pole, 1=Ball-Beam, 2=Incline, 3=Drone, 4=Gantry Crane

// ========================================================
// 1. CART-POLE VARIABLES
// ========================================================
float cp_x = 0, cp_v = 0;           
float cp_theta = 0.1, cp_omega = 0; 
float cp_M = 1.0, cp_m = 0.1, cp_l = 1.0, cp_g = 9.81;
float cp_mu_c = 0.05, cp_mu_p = 0.001, cp_appliedForce = 0; 
float cp_fuzzySensitivity = 0.3; 

// ========================================================
// 2. BALL-BEAM VARIABLES
// ========================================================
float bb_x = 0.5, bb_v = 0, bb_theta = 0, bb_targetTheta = 0;        
float bb_beamLength = 2.0, bb_g = 9.81, bb_ballRadius = 0.08, bb_servoSpeed = 8.0;       
float bb_fuzzySensitivity = 0.3; 

// ========================================================
// 3. TWO-WHEEL ROBOT (INCLINE) VARIABLES
// ========================================================
float rw_x = 0, rw_v = 0, rw_theta = 0.1, rw_omega = 0; 
float rw_M = 1.0, rw_m = 0.3, rw_l = 0.8, rw_g = 9.81;
float rw_planeAngle = radians(12), rw_appliedForce = 0; 
float rw_fuzzySensitivity = 0.3; 

// ========================================================
// 4. 2D DRONE (HYBRID FUZZY-PID) VARIABLES
// ========================================================
float dr_x = 0, dr_y = 0, dr_vx = 0, dr_vy = 0;
float dr_theta = 0.2, dr_omega = 0; // Roll angle and rate
float dr_m = 1.2, dr_I = 0.05, dr_d = 0.2, dr_g = 9.81;
float dr_TL = 0, dr_TR = 0; // Left and Right Thrust
float dr_fuzzySensitivity = 0.4; // Controls Fuzzy-PID Gain Scaling

// ========================================================
// 5. GANTRY CRANE (ANTI-SWAY FUZZY-PID) VARIABLES
// ========================================================
float gc_x = -1.5, gc_v = 0; // Cart position (starts off-center to show PID target seeking)
float gc_theta = 0, gc_omega = 0; // Payload swing
float gc_M = 2.0, gc_m = 0.5, gc_L = 1.5, gc_g = 9.81;
float gc_appliedForce = 0;
float gc_fuzzySensitivity = 0.3;


// ========================================================
// SETUP & MAIN DRAW LOOP
// ========================================================
void setup() {
  fullScreen(P2D);
  orientation(LANDSCAPE);
  textAlign(CENTER, CENTER);
  rectMode(CENTER);
}

void draw() {
  background(20, 25, 30); 
  
  int physicsSteps = 10;
  float dt = 0.016 / physicsSteps; 
  
  pushMatrix();
  translate(width / 2.0, height - 200); // Default origin for most sims
  
  if (currentSim == 0) {
    for (int i = 0; i < physicsSteps; i++) { cp_computeFuzzyLogic(); cp_updatePhysics(dt); }
    cp_drawEnvironment(); cp_drawCartAndPole(); cp_drawForceVectors();
  } else if (currentSim == 1) {
    translate(0, -50);
    for (int i = 0; i < physicsSteps; i++) { bb_computeFuzzyLogic(); bb_updatePhysics(dt); }
    bb_drawEnvironment(); bb_drawBeamAndBall(); bb_drawForceVectors();
  } else if (currentSim == 2) {
    translate(0, 50);
    for (int i = 0; i < physicsSteps; i++) { rw_computeFuzzyLogic(); rw_updatePhysics(dt); }
    rw_drawEnvironment(); rw_drawRobot(); rw_drawForceVectors();
  } else if (currentSim == 3) {
    translate(0, -100);
    for (int i = 0; i < physicsSteps; i++) { dr_computeFuzzyPID(); dr_updatePhysics(dt); }
    dr_drawEnvironment(); dr_drawDrone(); dr_drawForceVectors();
  } else if (currentSim == 4) {
    translate(0, -250); // Move crane track to top of screen
    for (int i = 0; i < physicsSteps; i++) { gc_computeHybridControl(); gc_updatePhysics(dt); }
    gc_drawEnvironment(); gc_drawCrane(); gc_drawForceVectors();
  }
  popMatrix();
  
  // Draw UI Overlays
  if (currentSim == 0) drawHUD("CART-POLE: PURE FUZZY", "DRAG CENTER (Disturb Pole)", cp_fuzzySensitivity);
  else if (currentSim == 1) drawHUD("BALL-BEAM: PURE FUZZY", "DRAG CENTER (Shove Ball)", bb_fuzzySensitivity);
  else if (currentSim == 2) drawHUD("INCLINE 2-WHEEL: PURE FUZZY", "DRAG CENTER (Disturb Robot)", rw_fuzzySensitivity);
  else if (currentSim == 3) drawHUD("DRONE: FUZZY-TUNED PID (ATTITUDE)", "DRAG CENTER (Wind Gust)", dr_fuzzySensitivity);
  else if (currentSim == 4) drawHUD("GANTRY CRANE: HYBRID PID + FUZZY ANTI-SWAY", "DRAG CENTER (Shove Payload)", gc_fuzzySensitivity);
  
  drawNextButton();
}


// ========================================================
// 1. CART-POLE IMPLEMENTATION (PURE FUZZY)
// ========================================================
void cp_computeFuzzyLogic() {
  float fN_A = triMF(cp_theta, -1.0, -cp_fuzzySensitivity, 0);
  float fZ_A = triMF(cp_theta, -cp_fuzzySensitivity, 0, cp_fuzzySensitivity);
  float fP_A = triMF(cp_theta, 0, cp_fuzzySensitivity, 1.0);
  float fN_V = triMF(cp_omega, -3.0, -1.0, 0);
  float fZ_V = triMF(cp_omega, -1.0, 0, 1.0);
  float fP_V = triMF(cp_omega, 0, 1.0, 3.0);

  float num = 0, den = 0;
  float[] w = {
    min(fN_A, fN_V), min(fN_A, fZ_V), min(fN_A, fP_V),
    min(fZ_A, fN_V), min(fZ_A, fZ_V), min(fZ_A, fP_V),
    min(fP_A, fN_V), min(fP_A, fZ_V), min(fP_A, fP_V)
  };
  float[] f = {-50, -20, 0, -20, 0, 20, 0, 20, 50};
  
  for(int i=0; i<9; i++) { num += w[i]*f[i]; den += w[i]; }
  
  float positionCorrection = -cp_x * 3.0 - cp_v * 2.0; 
  cp_appliedForce = constrain((den==0?0:(num/den)) + positionCorrection, -80, 80);
}

void cp_updatePhysics(float dt) {
  float sinT = sin(cp_theta), cosT = cos(cp_theta);
  float fCart = cp_mu_c * Math.signum(cp_v), fPole = (cp_mu_p * cp_omega) / (cp_m * cp_l);
  float temp = (cp_appliedForce + cp_m * cp_l * cp_omega * cp_omega * sinT - fCart) / (cp_M + cp_m);
  float alpha = (cp_g * sinT - cosT * temp - fPole) / (cp_l * (4.0/3.0 - (cp_m * cosT * cosT) / (cp_M + cp_m)));
  float acc = temp - (cp_m * cp_l * alpha * cosT) / (cp_M + cp_m);

  cp_omega += alpha * dt; cp_theta += cp_omega * dt;
  cp_v += acc * dt; cp_x += cp_v * dt;
  if (abs(cp_theta) > HALF_PI) { cp_theta = 0.1; cp_omega = 0; cp_x = 0; cp_v = 0; }
}

void cp_drawEnvironment() { stroke(100); strokeWeight(4); line(-width, 20, width, 20); }
void cp_drawCartAndPole() {
  float sPx = 100, cX = cp_x * sPx;
  fill(50, 120, 200); noStroke(); rect(cX, 0, 100, 40, 5);
  fill(150); ellipse(cX - 30, 20, 20, 20); ellipse(cX + 30, 20, 20, 20);
  float pX = cX + sin(cp_theta) * (cp_l * 2 * sPx), pY = -cos(cp_theta) * (cp_l * 2 * sPx);
  stroke(200, 200, 50); strokeWeight(8); line(cX, 0, pX, pY);
  fill(255); noStroke(); ellipse(cX, 0, 15, 15);
  fill(255, 100, 100); ellipse(cX + sin(cp_theta)*cp_l*sPx, -cos(cp_theta)*cp_l*sPx, 24, 24);
}
void cp_drawForceVectors() {
  float sPx = 100, cX = cp_x * sPx;
  drawArrow(cX, 0, cX + cp_appliedForce * 2.5, 0, color(50, 255, 50));
}


// ========================================================
// 2. BALL-BEAM IMPLEMENTATION (PURE FUZZY)
// ========================================================
void bb_computeFuzzyLogic() {
  float fN_P = triMF(bb_x, -1.0, -bb_fuzzySensitivity, 0);
  float fZ_P = triMF(bb_x, -bb_fuzzySensitivity, 0, bb_fuzzySensitivity);
  float fP_P = triMF(bb_x, 0, bb_fuzzySensitivity, 1.0);
  float fN_V = triMF(bb_v, -2.0, -0.5, 0);
  float fZ_V = triMF(bb_v, -0.5, 0, 0.5);
  float fP_V = triMF(bb_v, 0, 0.5, 2.0);

  float num = 0, den = 0;
  float[] w = { min(fN_P, fN_V), min(fN_P, fZ_V), min(fN_P, fP_V), min(fZ_P, fN_V), min(fZ_P, fZ_V), min(fZ_P, fP_V), min(fP_P, fN_V), min(fP_P, fZ_V), min(fP_P, fP_V) };
  float[] f = { -0.4, -0.15, 0, -0.15, 0, 0.15, 0, 0.15, 0.4 };
  
  for(int i=0; i<9; i++) { num += w[i]*f[i]; den += w[i]; }
  bb_targetTheta = constrain((den==0?0:(num/den)), -0.6, 0.6); 
}

void bb_updatePhysics(float dt) {
  bb_theta += (bb_targetTheta - bb_theta) * bb_servoSpeed * dt;
  bb_v += (-(5.0 / 7.0) * bb_g * sin(bb_theta) - 1.5 * bb_v) * dt;
  bb_x += bb_v * dt;
  if (abs(bb_x) > (bb_beamLength / 2.0) + bb_ballRadius) { bb_x = 0; bb_v = 0; bb_theta = 0; }
}

void bb_drawEnvironment() {
  stroke(100); strokeWeight(2); line(0, 0, 0, 150); 
  fill(80); triangle(0, 0, -20, 150, 20, 150); ellipse(0, 0, 20, 20); 
}
void bb_drawBeamAndBall() {
  float sPx = 250; 
  pushMatrix(); rotate(-bb_theta); 
  stroke(200, 200, 50); strokeWeight(12); line(-bb_beamLength/2 * sPx, 0, bb_beamLength/2 * sPx, 0);
  pushMatrix(); rotate(-bb_targetTheta + bb_theta); stroke(255, 255, 255, 50); strokeWeight(4); line(-bb_beamLength/2 * sPx, 0, bb_beamLength/2 * sPx, 0); popMatrix();
  fill(50, 150, 255); noStroke(); ellipse(bb_x * sPx, -bb_ballRadius * sPx - 6, bb_ballRadius * 2 * sPx, bb_ballRadius * 2 * sPx); 
  popMatrix();
}
void bb_drawForceVectors() {
  float sPx = 250, bR = bb_ballRadius * sPx;
  float bX = (bb_x * sPx) * cos(-bb_theta) - (-bR - 6) * sin(-bb_theta);
  float bY = (bb_x * sPx) * sin(-bb_theta) + (-bR - 6) * cos(-bb_theta);
  drawArrow(bX, bY, bX, bY + 80, color(255, 50, 50)); // Gravity
}


// ========================================================
// 3. TWO-WHEEL INCLINE ROBOT IMPLEMENTATION
// ========================================================
void rw_computeFuzzyLogic() {
  float t_abs = rw_theta + rw_planeAngle; 
  float fN_A = triMF(t_abs, -1.0, -rw_fuzzySensitivity, 0), fZ_A = triMF(t_abs, -rw_fuzzySensitivity, 0, rw_fuzzySensitivity), fP_A = triMF(t_abs, 0, rw_fuzzySensitivity, 1.0);
  float fN_V = triMF(rw_omega, -3.0, -1.0, 0), fZ_V = triMF(rw_omega, -1.0, 0, 1.0), fP_V = triMF(rw_omega, 0, 1.0, 3.0);

  float num = 0, den = 0;
  float[] w = { min(fN_A, fN_V), min(fN_A, fZ_V), min(fN_A, fP_V), min(fZ_A, fN_V), min(fZ_A, fZ_V), min(fZ_A, fP_V), min(fP_A, fN_V), min(fP_A, fZ_V), min(fP_A, fP_V) };
  float[] f = {-80, -30, 0, -30, 0, 30, 0, 30, 80};
  
  for(int i=0; i<9; i++) { num += w[i]*f[i]; den += w[i]; }
  rw_appliedForce = constrain((den==0?0:(num/den)) - rw_x * 8.0 - rw_v * 5.0, -100, 100);
}

void rw_updatePhysics(float dt) {
  float t_abs = rw_theta + rw_planeAngle, sinAbs = sin(t_abs), cosLocal = cos(rw_theta);
  float temp = (rw_appliedForce - (rw_M + rw_m)*rw_g*sin(rw_planeAngle) + rw_m*rw_l*rw_omega*rw_omega*sin(rw_theta) - 0.05*Math.signum(rw_v)) / (rw_M + rw_m);
  float alpha = (rw_g * sinAbs - cosLocal * temp) / (rw_l * (4.0/3.0 - (rw_m*cosLocal*cosLocal)/(rw_M+rw_m)));
  
  rw_omega += alpha * dt; rw_theta += rw_omega * dt;
  rw_v += (temp - (rw_m * rw_l * alpha * cosLocal) / (rw_M + rw_m)) * dt; rw_x += rw_v * dt;
  if (abs(t_abs) > HALF_PI) { rw_theta = 0.1; rw_omega = 0; rw_x = 0; rw_v = 0; }
}

void rw_drawEnvironment() { pushMatrix(); rotate(-rw_planeAngle); stroke(100); strokeWeight(4); line(-width, 0, width, 0); popMatrix(); }
void rw_drawRobot() {
  float sPx = 100, wR = 25;
  pushMatrix(); rotate(-rw_planeAngle); translate(rw_x * sPx, -wR); 
  pushMatrix(); rotate(rw_theta); stroke(50, 180, 200); strokeWeight(12); line(0, 0, 0, -rw_l * 2 * sPx); fill(255, 100, 100); noStroke(); ellipse(0, -rw_l * sPx, 24, 24); popMatrix();
  fill(50); stroke(200); strokeWeight(2); ellipse(0, 0, wR*2, wR*2); fill(255); ellipse(0, 0, 8, 8); 
  popMatrix();
}
void rw_drawForceVectors() {
  pushMatrix(); rotate(-rw_planeAngle); 
  drawArrow(rw_x * 100, -25, rw_x * 100 + rw_appliedForce * 1.5, -25, color(50, 255, 50));
  popMatrix();
}


// ========================================================
// 4. 2D DRONE (FUZZY-TUNED PID CONTROL)
// ========================================================
void dr_computeFuzzyPID() {
  // 1. Base PID for Altitude (Maintains target Y = 0)
  float targetY = 0;
  float Kp_y = 15.0, Kd_y = 5.0;
  float altForce = Kp_y * (targetY - dr_y) - Kd_y * dr_vy + (dr_m * dr_g); // Include gravity feed-forward
  
  // 2. Fuzzy Logic dictates the Gain Tuning for Roll
  // If drone is heavily tilted, increase Kp significantly to snap back. If near level, keep Kp low to prevent oscillation.
  float err = 0 - dr_theta;
  float fuzzyHigh = 1.0 - triMF(err, -dr_fuzzySensitivity, 0, dr_fuzzySensitivity); // High when far from 0
  
  float baseKp = 5.0;
  float Kp_roll = baseKp + (fuzzyHigh * 15.0); // Boost Kp dynamically up to 20 based on fuzzy evaluation
  float Kd_roll = 1.5;
  
  float rollForce = Kp_roll * err - Kd_roll * dr_omega;
  
  // 3. Mixer: Combine Altitude and Roll requests into Left/Right motor thrusts
  dr_TL = constrain((altForce / 2.0) - rollForce, 0, 30); // Motors can't thrust negative
  dr_TR = constrain((altForce / 2.0) + rollForce, 0, 30);
  
  // Positional PID to stop drift
  dr_theta += (-dr_x * 0.05 - dr_vx * 0.05) * 0.01; // Lean into horizontal drift
}

void dr_updatePhysics(float dt) {
  // Linear acceleration (Note: Processing Y is down, so thrust pulls UP/negative Y)
  float totalThrust = dr_TL + dr_TR;
  float ax = (totalThrust * sin(dr_theta)) / dr_m;
  float ay = dr_g - (totalThrust * cos(dr_theta)) / dr_m; 
  
  // Angular acceleration
  float alpha = ((dr_TR - dr_TL) * dr_d) / dr_I;
  
  // Integrate
  dr_vx += ax * dt; dr_x += dr_vx * dt;
  dr_vy += ay * dt; dr_y += dr_vy * dt;
  dr_omega += alpha * dt; dr_theta += dr_omega * dt;
  
  // Floor collision
  if (dr_y > 2.0) { dr_y = 2.0; dr_vy = 0; dr_omega = 0; }
}

void dr_drawEnvironment() {
  stroke(100); strokeWeight(4); line(-width, 200, width, 200); // Floor
  fill(50, 150); noStroke(); rect(0, 0, 400, 100, 20); // Target zone
}

void dr_drawDrone() {
  float sPx = 100;
  pushMatrix();
  translate(dr_x * sPx, dr_y * sPx);
  rotate(dr_theta);
  
  // Chassis
  stroke(200); strokeWeight(6);
  line(-dr_d * sPx, 0, dr_d * sPx, 0); // Arms
  fill(100, 150, 255); noStroke(); ellipse(0, 0, 30, 20); // Body
  
  // Props (visual spinning based on thrust)
  fill(255, 100); noStroke();
  ellipse(-dr_d * sPx, -5, dr_TL * 3 + 10, 8); // Left Prop
  ellipse(dr_d * sPx, -5, dr_TR * 3 + 10, 8);  // Right Prop
  
  popMatrix();
}

void dr_drawForceVectors() {
  float sPx = 100;
  pushMatrix();
  translate(dr_x * sPx, dr_y * sPx); rotate(dr_theta);
  drawArrow(-dr_d * sPx, 0, -dr_d * sPx, -dr_TL * 2, color(50, 255, 50));
  drawArrow(dr_d * sPx, 0, dr_d * sPx, -dr_TR * 2, color(50, 255, 50));
  popMatrix();
}


// ========================================================
// 5. GANTRY CRANE (HYBRID PID + FUZZY ANTI-SWAY)
// ========================================================
void gc_computeHybridControl() {
  // 1. Standard PID Controller for moving the Cart to Target (X = 0)
  float targetX = 0;
  float Kp_x = 10.0, Kd_x = 5.0;
  float pidForce = Kp_x * (targetX - gc_x) - Kd_x * gc_v;
  
  // 2. Fuzzy Logic Controller for Anti-Sway (Eliminate Payload Swing)
  float fN_A = triMF(gc_theta, -1.0, -gc_fuzzySensitivity, 0);
  float fZ_A = triMF(gc_theta, -gc_fuzzySensitivity, 0, gc_fuzzySensitivity);
  float fP_A = triMF(gc_theta, 0, gc_fuzzySensitivity, 1.0);
  float fN_V = triMF(gc_omega, -2.0, -0.5, 0);
  float fZ_V = triMF(gc_omega, -0.5, 0, 0.5);
  float fP_V = triMF(gc_omega, 0, 0.5, 2.0);

  float num = 0, den = 0;
  float[] w = { min(fN_A, fN_V), min(fN_A, fZ_V), min(fN_A, fP_V), min(fZ_A, fN_V), min(fZ_A, fZ_V), min(fZ_A, fP_V), min(fP_A, fN_V), min(fP_A, fZ_V), min(fP_A, fP_V) };
  float[] f = {-30, -15, 0, -15, 0, 15, 0, 15, 30}; // Forces needed to 'catch' the swing
  
  for(int i=0; i<9; i++) { num += w[i]*f[i]; den += w[i]; }
  float fuzzySwayForce = (den == 0) ? 0 : (num / den);
  
  // 3. Hybrid Combination: Move cart, but prioritize zeroing the sway.
  gc_appliedForce = constrain(pidForce + fuzzySwayForce, -60, 60);
}

void gc_updatePhysics(float dt) {
  float sinT = sin(gc_theta), cosT = cos(gc_theta);
  
  // Non-linear equations of motion for overhead crane
  float temp = (gc_appliedForce + gc_m * gc_L * gc_omega * gc_omega * sinT - 2.0 * gc_v) / (gc_M + gc_m);
  float alpha = (-gc_g * sinT - cosT * temp) / (gc_L * (1.0 - (gc_m * cosT * cosT) / (gc_M + gc_m)));
  float acc = temp - (gc_m * gc_L * alpha * cosT) / (gc_M + gc_m);

  gc_omega += alpha * dt; gc_theta += gc_omega * dt;
  gc_v += acc * dt; gc_x += gc_v * dt;
}

void gc_drawEnvironment() {
  stroke(100); strokeWeight(10); line(-width, 0, width, 0); // Ceiling Rail
  fill(50, 150); noStroke(); rect(0, 50, 150, 20, 5); // Target Zone Marker
}

  float sPx = 150, cX = gc_x * sPx;
  
  // Cart
  fill(255, 150, 50); noStroke(); rect(cX, 20, 80, 40, 8);
  fill(50); ellipse(cX - 25, 0, 16, 16); ellipse(cX + 25, 0, 16, 16); // Wheels
  
  // Cable & Payload
  float pX = cX + sin(gc_theta) * (gc_L * sPx);
  float pY = 20 + cos(gc_theta) * (gc_L * sPx); // Processing Y is down
  
  stroke(200); strokeWeight(3); line(cX, 40, pX, pY);
  fill(100, 200, 255); noStroke(); rect(pX, pY, 50, 50, 5); // Heavy crate
}

void gc_drawForceVectors() {
  float sPx = 150, cX = gc_x * sPx;
  drawArrow(cX, 20, cX + gc_appliedForce * 2, 20, color(50, 255, 50));
}


// ========================================================
// SHARED HELPER FUNCTIONS
// ========================================================
void drawHUD(String title, String hint, float sens) {
  fill(0, 150); stroke(100); strokeWeight(2); rectMode(CORNER);
  rect(20, 20, 350, 120, 10);
  fill(255); textSize(14); textAlign(LEFT, TOP); text(title, 30, 30);
  
  // Shared Membership/Gain visualization
  float axX = 30, axY = 110, axW = 320, axH = 50;
  stroke(200); strokeWeight(1); line(axX, axY, axX + axW, axY); 
  float center = axX + axW/2; float sensPx = sens * axW/2;
  
  noFill(); strokeWeight(2);
  stroke(255, 50, 50); triangle(axX, axY, axX, axY - axH, center, axY); // Negative
  stroke(50, 255, 50); triangle(center - sensPx, axY, center, axY - axH, center + sensPx, axY); // Zero (Tweakable)
  stroke(50, 150, 255); triangle(center, axY, axX + axW, axY - axH, axX + axW, axY); // Positive
  
  rectMode(CENTER);
  fill(0, 180); noStroke(); rect(width/2, height - 25, width, 50);
  fill(255); textSize(18); textAlign(CENTER, CENTER);
  text("TAP LEFT (Widen Gain) | TAP RIGHT (Narrow Gain) | " + hint, width/2, height - 25);
}

float triMF(float val, float a, float b, float c) {
  if (val <= a || val >= c) return 0;
  if (val == b) return 1;
  if (val < b) return (val - a) / (b - a);
  return (c - val) / (c - b);
}

void drawArrow(float x1, float y1, float x2, float y2, int c) {
  if (dist(x1, y1, x2, y2) < 2) return; 
  stroke(c); fill(c); strokeWeight(3); line(x1, y1, x2, y2);
  pushMatrix(); translate(x2, y2); rotate(atan2(y2 - y1, x2 - x1)); 
  noStroke(); triangle(0, 0, -10, 5, -10, -5); popMatrix();
}

void drawNextButton() {
  rectMode(CORNER); fill(200, 50, 50); stroke(255); strokeWeight(2);
  rect(width - 160, 20, 140, 50, 8); 
  fill(255); textSize(18); textAlign(CENTER, CENTER); text("NEXT SIM \u2192", width - 90, 45); 
  rectMode(CENTER);
}

// ========================================================
// INTERACTION LOGIC
// ========================================================
void mousePressed() {
  if (mouseX > width - 160 && mouseX < width - 20 && mouseY > 20 && mouseY < 70) {
    currentSim = (currentSim + 1) % 5; // Cycle through 5 sims
    return; 
  }
  if (mouseY > height - 100) return; 
  
  float adj = (mouseX < width / 3) ? 0.05 : ((mouseX > width * 0.66) ? -0.05 : 0);
  if(adj != 0) {
    if (currentSim == 0) cp_fuzzySensitivity = constrain(cp_fuzzySensitivity + adj, 0.1, 0.9); 
    else if (currentSim == 1) bb_fuzzySensitivity = constrain(bb_fuzzySensitivity + adj, 0.1, 0.9); 
    else if (currentSim == 2) rw_fuzzySensitivity = constrain(rw_fuzzySensitivity + adj, 0.1, 0.9); 
    else if (currentSim == 3) dr_fuzzySensitivity = constrain(dr_fuzzySensitivity + adj, 0.1, 0.9); 
    else if (currentSim == 4) gc_fuzzySensitivity = constrain(gc_fuzzySensitivity + adj, 0.1, 0.9); 
  }
}

void mouseDragged() {
  if (mouseX > width - 160 && mouseX < width - 20 && mouseY > 20 && mouseY < 70) return;
  if (mouseX > width / 3 && mouseX < width * 0.66) {
    float dist = (mouseX - pmouseX);
    if (currentSim == 0) cp_omega += dist * 0.1; 
    else if (currentSim == 1) bb_v += dist * 0.05; 
    else if (currentSim == 2) rw_omega += dist * 0.1; 
    else if (currentSim == 3) dr_omega += dist * 0.2; // Wind gust hitting drone roll
    else if (currentSim == 4) gc_omega += dist * 0.05; // Shove crane payload
  }
}
